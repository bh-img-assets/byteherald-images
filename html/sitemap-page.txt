(function () {
  var box = document.getElementById('bh-sitemap');
  if (!box) { return; }
  var PER = 150;
  var all = [];
  function fetchPage(start) {
    var url = 'https://www.byteherald.com/feeds/posts/default?alt=json&max-results=' + PER + '&start-index=' + start;
    return fetch(url).then(function (r) { if (!r.ok) { throw new Error('feed ' + r.status); } return r.json(); });
  }
  function collect(feed) {
    var entries = (feed && feed.feed && feed.feed.entry) || [];
    entries.forEach(function (e) {
      var title = (e.title && e.title.$t) || '';
      var link = '';
      (e.link || []).forEach(function (l) { if (l.rel === 'alternate') { link = l.href; } });
      if (title && link) { all.push({ title: title, url: link }); }
    });
    return entries.length;
  }
  function load(start) {
    return fetchPage(start).then(function (feed) {
      var n = collect(feed);
      if (n === PER) { return load(start + PER); }
    });
  }
  function letterOf(title) {
    var c = title.charAt(0).toUpperCase();
    return (c >= 'A' && c <= 'Z') ? c : '#';
  }
  function merge() {
    var seen = {};
    Array.prototype.forEach.call(box.querySelectorAll('a'), function (a) { seen[a.href] = true; });
    var fresh = all.filter(function (p) { return !seen[p.url]; });
    if (fresh.length === 0) { return; }
    var groups = {};
    Array.prototype.forEach.call(box.querySelectorAll('h2'), function (h) {
      var ul = h.nextElementSibling;
      if (ul && ul.tagName === 'UL') { groups[h.textContent] = ul; }
    });
    fresh.forEach(function (p) {
      var L = letterOf(p.title);
      var ul = groups[L];
      if (!ul) {
        var h = document.createElement('h2');
        h.textContent = L;
        box.appendChild(h);
        ul = document.createElement('ul');
        box.appendChild(ul);
        groups[L] = ul;
      }
      var li = document.createElement('li');
      var a = document.createElement('a');
      a.href = p.url;
      a.textContent = p.title;
      li.appendChild(a);
      ul.appendChild(li);
    });
    Object.keys(groups).forEach(function (L) {
      var ul = groups[L];
      var items = Array.prototype.slice.call(ul.children);
      items.sort(function (x, y) { return x.textContent.toLowerCase().localeCompare(y.textContent.toLowerCase()); });
      items.forEach(function (li) { ul.appendChild(li); });
    });
    var count = document.getElementById('bh-count');
    if (count) { count.textContent = box.querySelectorAll('a').length + ' articles, A to Z. New articles appear here automatically.'; }
  }
  load(1).then(merge).catch(function () {});
})();
